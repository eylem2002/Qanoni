```typescript
import {
  Document, Packer, Paragraph, TextRun, AlignmentType,
  HeadingLevel, BorderStyle, TableRow, TableCell, Table,
  WidthType, ShadingType, Header, Footer, PageNumber,
} from "docx";

export interface WordDocumentOptions {
  contractText: string;
  contractTypeLabel: string;
  partyName: string;
  generatedDate: string;
}

export async function buildWordDocument(opts: WordDocumentOptions): Promise<Buffer> {
  const { contractText, contractTypeLabel, partyName, generatedDate } = opts;

  // Split contract text into paragraphs on newline
  const lines = contractText.split("\n");

  const contentParagraphs = lines.map((line) => {
    const trimmed = line.trim();

    // Section separators like ━━━━ become a thin rule
    if (/^[━─=]{5,}$/.test(trimmed)) {
      return new Paragraph({
        border: { bottom: { color: "C9A94E", size: 6, style: BorderStyle.SINGLE } },
        spacing: { before: 80, after: 80 },
      });
    }

    // Lines starting with "الجزء" or "البند" or "عقد عمل" are headings
    const isHeading =
      /^(الجزء|البند|عقد |اتفاقية |المراجع القانونية|التوقيعات)/.test(trimmed);

    return new Paragraph({
      alignment: AlignmentType.RIGHT,
      bidirectional: true,
      spacing: { before: isHeading ? 160 : 60, after: 40 },
      children: [
        new TextRun({
          text: trimmed || " ",
          bold: isHeading,
          size: isHeading ? 24 : 22,
          font: "Arial",
          rightToLeft: true,
          color: isHeading ? "1A1A1A" : "2C2C2C",

  });

  const doc = new Document({
    styles: {
      default: {
        document: {
          run: { font: "Arial", size: 22, rightToLeft: true },
          paragraph: { alignment: AlignmentType.RIGHT, bidirectional: true },
        },
      },
    },

    sections: [
      {
        properties: {
          page: {
            margin: { top: 1200, right: 1000, bottom: 1200, left: 1000 },
          },
        },

        headers: {
          default: new Header({
            children: [
              new Paragraph({
                alignment: AlignmentType.CENTER,
                bidirectional: true,
                border: {
                  bottom: { color: "C9A94E", size: 8, style: BorderStyle.SINGLE },
                },
                spacing: { after: 120 },
                children: [
                  new TextRun({
                    text: `قانوني AI — ${contractTypeLabel}`,
                    bold: true,
                    size: 22,
                    font: "Arial",
                    color: "C9A94E",

                  new TextRun({
                    text: `     |     تاريخ الإصدار: ${generatedDate}`,
                    size: 18,
                    font: "Arial",
                    color: "888888",
                  }),
                ],
              }),
            ],
          }),
        },

        footers: {
          default: new Footer({
            children: [
              new Paragraph({
                alignment: AlignmentType.CENTER,
                bidirectional: true,
                border: {
                  top: { color: "CCCCCC", size: 4, style: BorderStyle.SINGLE },
                },
                spacing: { before: 100 },
                children: [
                  new TextRun({
                    text: "هذا العقد مُولَّد بمساعدة الذكاء الاصطناعي استناداً للقانون الأردني. يُنصح بمراجعة محامٍ مرخص قبل التوقيع. — قانوني AI    |    صفحة ",
                    size: 16,
                    font: "Arial",
                    color: "888888",
                    rightToLeft: true,
                  }),
                  new TextRun({
                    children: [PageNumber.CURRENT],
                    size: 16,
                    font: "Arial",
                    color: "888888",
                  }),
                ],

        children: [

            bidirectional: true,
```

## TASK B — Build the Contract Reviewer (Feature 2)

### What it does
User uploads an existing contract (PDF, DOCX, or plain text).
The system extracts the text, sends it to Claude with a review prompt.
Claude returns a structured JSON list of clause findings — each with:
- `status`: "green" | "yellow" | "red"
- `clause`: the clause name in Arabic
- `finding`: what was found (in Arabic)
- `suggestion`: suggested fix or action (in Arabic)
- `severity`: "safe" | "unusual" | "illegal"

The frontend renders a risk report with colored cards.

---

### Step B1 — Install backend dependencies for file parsing

```bash
cd artifacts/api-server

npm install @types/multer @types/pdf-parse --save-dev
```

- `multer` — handles multipart/form-data file uploads in Express
- `pdf-parse` — extracts text from PDF files
- `mammoth` — extracts text from DOCX files (also works for .doc)

### Step B2 — Create the review route file

Create a new file: `artifacts/api-server/src/routes/contracts/review.ts`

```typescript
import { Router } from "express";
import multer from "multer";
import pdfParse from "pdf-parse";
import mammoth from "mammoth";
import { anthropic } from "@workspace/integrations-anthropic-ai";

const reviewRouter = Router();

// Store file in memory — no disk writes needed, file is processed and discarded
const upload = multer({
  storage: multer.memoryStorage(),
  limits: {
    fileSize: 5 * 1024 * 1024, // 5 MB max
    files: 1,
  },
  fileFilter: (_req, file, cb) => {
    const allowed = [
      "application/pdf",
      "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
      "application/msword",
      "text/plain",
    ];
    if (allowed.includes(file.mimetype)) {
      cb(null, true);
    } else {
      cb(new Error("نوع الملف غير مدعوم. المقبول: PDF, DOCX, TXT"));
    }
  },
});

// ─── Text extractor ───────────────────────────────────────────
async function extractText(buffer: Buffer, mimetype: string): Promise<string> {
  if (mimetype === "application/pdf") {
    const result = await pdfParse(buffer);
    return result.text;
  }

  if (
    mimetype === "application/vnd.openxmlformats-officedocument.wordprocessingml.document" ||
    mimetype === "application/msword"
  ) {
    const result = await mammoth.extractRawText({ buffer });
    return result.value;
  }

  if (mimetype === "text/plain") {
    return buffer.toString("utf-8");
  }

  throw new Error("نوع ملف غير مدعوم");
}

// ─── Review prompt ────────────────────────────────────────────
function buildReviewPrompt(contractText: string): string {
  return `أنت محامٍ أردني خبير في مراجعة العقود. مهمتك مراجعة العقد المرفق أدناه وتحديد المخاطر القانونية.

قواعد صارمة:
1. أجب فقط بـ JSON صحيح — لا نص قبله ولا بعده ولا backticks.
2. كل بند يجب تصنيفه إلى: "green" (آمن) أو "yellow" (غير معتاد / يحتاج مراجعة) أو "red" (مشكلة قانونية خطيرة أو بند مفقود إلزامي).
3. لا تخترع بنوداً غير موجودة في العقد.
4. إذا كان بند مفقود لكنه إلزامي قانوناً (مثل الراتب أو إنهاء العقد)، أضفه كـ red مع finding يبدأ بـ "بند مفقود:".
5. أعطِ على الأقل 6 بنود ولا تتجاوز 15 بنداً.
6. اكتب كل النصوص باللغة العربية.

أعد الرد بهذا الشكل تماماً (JSON array):
[
  {
    "status": "green" | "yellow" | "red",
    "severity": "safe" | "unusual" | "illegal",
    "clause": "اسم البند بالعربية",
    "finding": "ما وجدته في العقد — جملة أو جملتان",
    "suggestion": "الإجراء الموصى به — جملة واحدة. إذا كان آمناً اكتب: لا يلزم تغيير."
  }
]

══════════════════════════════════════════════════════
نص العقد للمراجعة:
══════════════════════════════════════════════════════
${contractText.slice(0, 12000)}
══════════════════════════════════════════════════════

أجب الآن بـ JSON فقط:`;
}

// ─── Route ────────────────────────────────────────────────────
reviewRouter.post(
  "/contracts/review",
  upload.single("contract"),
  async (req, res) => {
    if (!req.file) {
      res.status(400).json({ error: "لم يتم رفع أي ملف" });
      return;
    }

    let contractText: string;
    try {
      contractText = await extractText(req.file.buffer, req.file.mimetype);
    } catch (err) {
      res.status(422).json({ error: "تعذر قراءة الملف. تأكد أنه PDF أو DOCX أو TXT سليم." });
      return;
    }

    if (contractText.trim().length < 100) {
      res.status(422).json({ error: "الملف المرفوع يبدو فارغاً أو لا يحتوي على نص قابل للقراءة" });
      return;
    }

    try {
      const message = await anthropic.messages.create({
        model: "claude-sonnet-4-6",
        max_tokens: 4096,
        messages: [{ role: "user", content: buildReviewPrompt(contractText) }],
      });

      const rawText = message.content[0].type === "text" ? message.content[0].text : "";

      // Strip any accidental markdown backticks before parsing
      const cleaned = rawText.replace(/```json|```/gi, "").trim();

      let findings: unknown[];
      try {
        findings = JSON.parse(cleaned);
        if (!Array.isArray(findings)) throw new Error("not an array");
      } catch {
        req.log.error({ rawText }, "Failed to parse review JSON from Claude");
        res.status(500).json({ error: "تعذر تحليل نتيجة المراجعة. حاول مرة أخرى." });
        return;
      }

      res.json({
        findings,
        charCount: contractText.length,
        filename: req.file.originalname,
      });
    } catch (err) {
      req.log.error({ err }, "Contract review failed");
      res.status(500).json({ error: "فشل في مراجعة العقد. يرجى المحاولة مرة أخرى." });
    }
  }
);

export default reviewRouter;
```

```typescript
                  onChange={(e) => {
                    const file = e.target.files?.[0];
                    if (file) handleFileSelect(file);
                  }}
                />

                {selectedFile ? (
                  <div className="flex flex-col items-center gap-2">
                    <FileText className="w-10 h-10 text-green-400" />
                    <p className="text-sm font-medium text-green-400 break-all">{selectedFile.name}</p>
                    <p className="text-xs text-muted-foreground">
                      {(selectedFile.size / 1024).toFixed(1)} KB
                    </p>
                    <button
                      type="button"
                      onClick={(e) => { e.stopPropagation(); setSelectedFile(null); setResult(null); }}
                      className="mt-1 flex items-center gap-1 text-xs text-red-400 hover:text-red-300 transition-colors"
                    >
                      <X className="w-3 h-3" /> إزالة الملف
                    </button>
                  </div>
                ) : (
                  <div className="flex flex-col items-center gap-3">
                    <Upload className="w-10 h-10 text-muted-foreground/60" />
                    <div>
                      <p className="text-sm font-medium text-gray-300">اسحب الملف هنا أو انقر للاختيار</p>
                      <p className="text-xs text-muted-foreground mt-1">PDF, DOCX, TXT — حتى 5 MB</p>
                    </div>
                  </div>
                )}
              </div>

              {/* Review button */}
              <Button
                onClick={handleReview}
                disabled={!selectedFile || isLoading}
                className="w-full h-12 text-base font-bold"
              >
                {isLoading ? (
                  <><Spinner className="ml-3 w-5 h-5 text-primary-foreground" /> جاري تحليل العقد...</>
                ) : (
                  "مراجعة العقد"
                )}
              </Button>

              {/* Info box */}
              <div className="rounded-lg border border-blue-500/20 bg-blue-500/5 p-3">
                <p className="text-xs text-blue-300/80 leading-relaxed">
                  يقوم الذكاء الاصطناعي بمراجعة البنود وتصنيفها إلى: آمن (أخضر)، يحتاج مراجعة (أصفر)، أو خطر قانوني (أحمر). هذه المراجعة لأغراض إرشادية فقط — لا تُغني عن محامٍ مرخص.
                </p>
              </div>

            </CardContent>
          </Card>
        </div>

        {/* Results Panel */}
        <div className="lg:col-span-7">
          <Card className="h-full border-border/50 bg-card/40 backdrop-blur-xl shadow-2xl flex flex-col rounded-2xl min-h-[500px]">
            <CardHeader className="border-b border-border/50 bg-secondary/30 py-4 shrink-0">
              <CardTitle className="text-lg font-bold text-white flex items-center gap-2">
                <FileText className="w-5 h-5 text-primary" />
                تقرير المخاطر
              </CardTitle>
            </CardHeader>

            {/* Summary bar + filter */}
            {counts && (
              <div className="border-b border-border/50 px-5 py-3 flex items-center gap-3 flex-wrap bg-secondary/10 shrink-0">
                <span className="text-xs text-muted-foreground ml-1">تصفية:</span>
                {(["all", "red", "yellow", "green"] as const).map((f) => {
                  const label = f === "all" ? `الكل (${result!.findings.length})` :
                    f === "red" ? `خطر (${counts.red})` :
                    f === "yellow" ? `مراجعة (${counts.yellow})` :
                    `آمن (${counts.green})`;
                  const colors = f === "red" ? "border-red-500/40 text-red-400 data-[active=true]:bg-red-500/15" :
                    f === "yellow" ? "border-amber-500/40 text-amber-400 data-[active=true]:bg-amber-500/15" :
                    f === "green" ? "border-green-500/40 text-green-400 data-[active=true]:bg-green-500/15" :
                    "border-border/60 text-muted-foreground data-[active=true]:bg-secondary";
                  return (
                    <button
                      key={f}
                      data-active={filter === f}
                      onClick={() => setFilter(f)}
                      className={`text-xs px-3 py-1 rounded-full border transition-all ${colors}`}
                    >
                      {label}
                    </button>
                  );
                })}
              </div>
            )}

            <CardContent className="flex-1 p-0 overflow-hidden">
              {isLoading && (
                <div className="h-full flex flex-col items-center justify-center gap-4 p-8">
                  <div className="relative">
                    <div className="w-16 h-16 rounded-full border-4 border-primary/20 border-t-primary animate-spin" />
                    <Scale className="w-6 h-6 text-primary absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 opacity-50" />
                  </div>
                  <div className="text-center">
                    <p className="font-bold text-white">جاري تحليل بنود العقد...</p>
                    <p className="text-sm text-muted-foreground mt-1">يتم فحص كل بند مقابل القانون الأردني</p>
                  </div>
                </div>
              )}

              {!isLoading && !result && (
                <div className="h-full flex flex-col items-center justify-center text-center p-8 opacity-70">
                  <div className="bg-secondary/40 p-5 rounded-full mb-4 border border-border/50">
                    <FileText className="w-12 h-12 text-muted-foreground/60" />
                  </div>
                  <p className="font-bold text-gray-300">ارفع عقداً لتبدأ المراجعة</p>
                  <p className="text-sm text-muted-foreground mt-2 max-w-xs">
                    يدعم النظام أي عقد أردني — عمل، إيجار، خدمات، توريد، أو غير ذلك
                  </p>
                </div>
              )}

              {!isLoading && result && (
                <div className="h-full overflow-y-auto p-5 space-y-3">
                  {filteredFindings.length === 0 ? (
                    <p className="text-center text-sm text-muted-foreground py-8">لا توجد نتائج لهذا التصنيف</p>
                  ) : (
                    filteredFindings.map((f, i) => {
                      const cfg = STATUS_CONFIG[f.status];
                      const Icon = cfg.icon;
                      return (
                        <div
                          key={i}
                          className={`rounded-xl border p-4 ${cfg.bg} ${cfg.border}`}
                        >
                          <div className="flex items-start justify-between gap-3 mb-2">
                            <div className="flex items-center gap-2">
                              <Icon className={`w-4 h-4 shrink-0 ${cfg.iconColor}`} />
                              <span className="text-sm font-bold text-white">{f.clause}</span>
                            </div>
                            <span className={`text-xs px-2 py-0.5 rounded-full font-medium shrink-0 ${cfg.labelBg}`}>
                              {cfg.label}
                            </span>
                          </div>
                          <p className="text-xs text-gray-300 leading-relaxed mb-2">{f.finding}</p>
                          {f.suggestion && f.suggestion !== "لا يلزم تغيير." && (
                            <div className="border-t border-white/10 pt-2 mt-2">
                              <p className="text-xs text-muted-foreground">
                                <span className="text-primary/80 font-medium">التوصية: </span>
                                {f.suggestion}
                              </p>
                            </div>
                          )}
                        </div>
                      );
                    })
                  )}
                </div>
              )}
            </CardContent>
          </Card>
        </div>
      </main>
    </div>
  );
}
```
